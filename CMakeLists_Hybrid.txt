cmake_minimum_required(VERSION 3.20)
project(VulkanPhysicsHybridBenchmark CXX)

set(CMAKE_CXX_STANDARD 20)
set(CMAKE_CXX_STANDARD_REQUIRED ON)

include(FetchContent)
FetchContent_Declare(
    highway
    GIT_REPOSITORY https://github.com/google/highway.git
    GIT_TAG        1.1.0
)
set(HWY_ENABLE_EXAMPLES OFF CACHE BOOL "" FORCE)
set(HWY_ENABLE_TESTS OFF CACHE BOOL "" FORCE)
set(HWY_ENABLE_CONTRIB OFF CACHE BOOL "" FORCE)
FetchContent_MakeAvailable(highway)

find_package(Threads REQUIRED)

add_executable(vulkan_physics_hybrid_benchmark vulkan_physics_hybrid_benchmark.cpp)
target_link_libraries(vulkan_physics_hybrid_benchmark PRIVATE hwy Threads::Threads)

if (MSVC)
    target_compile_options(vulkan_physics_hybrid_benchmark PRIVATE /O2 /arch:AVX2)
else()
    target_compile_options(vulkan_physics_hybrid_benchmark PRIVATE -O3 -mavx2)
endif()
